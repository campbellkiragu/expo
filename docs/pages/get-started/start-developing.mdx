import React, { useEffect, useState, useContext } from 'react';
import { View, Text, TouchableOpacity, Image } from 'react-native';
import TrackPlayer, { usePlaybackState } from 'react-native-track-player';
import { fetchMusicRecommendations } from '../services/fetchMusic';
import { getUserPreferences } from '../services/cacheService';
import { ThemeContext } from '../context/ThemeContext';

export default function MusicPlayer() {
    const [tracks, setTracks] = useState([]);
    const [currentTrackIndex, setCurrentTrackIndex] = useState(0);
    const [isShuffled, setIsShuffled] = useState(false);
    const playbackState = usePlaybackState();
    const { theme } = useContext(ThemeContext);

    useEffect(() => {
        async function loadMusic() {
            const userPrefs = await getUserPreferences();
            const mood = "Happy"; 
            const music = await fetchMusicRecommendations(mood, userPrefs.selectedArtists || []);
            setTracks(music);
            await TrackPlayer.add(music);
        }
        loadMusic();
    }, []);

    return (
        <View style={{ backgroundColor: theme.background, padding: 20 }}>
            {tracks.length > 0 && (
                <>
                    <Image source={{ uri: tracks[currentTrackIndex].artwork }} style={{ width: 200, height: 200 }} />
                    <Text style={{ color: theme.text }}>{tracks[currentTrackIndex].title}</Text>
                    <View>
                        <TouchableOpacity onPress={() => TrackPlayer.skipToPrevious()}><Text>⏮️</Text></TouchableOpacity>
                        <TouchableOpacity onPress={() => TrackPlayer.play()}><Text>{playbackState === 'playing' ? "⏸️" : "▶️"}</Text></TouchableOpacity>
                        <TouchableOpacity onPress={() => TrackPlayer.skipToNext()}><Text>⏭️</Text></TouchableOpacity>
                        <TouchableOpacity onPress={() => setIsShuffled(!isShuffled)}><Text>{isShuffled ? "🔀" : "🔂"}</Text></TouchableOpacity>
                    </View>
                </>
            )}
        </View>
    );
}
export const fetchMusicRecommendations = async (mood, favoriteArtists) => {
    const response = await fetch(`https://api.spotify.com/v1/recommendations?seed_artists=${favoriteArtists.join(',')}&target_valence=${mood === 'Happy' ? 0.9 : 0.3}`, {
        headers: { Authorization: `Bearer YOUR_SPOTIFY_ACCESS_TOKEN` }
    });
    const data = await response.json();
    return data.tracks;
};
import AsyncStorage from '@react-native-async-storage/async-storage';

export const storeUserPreferences = async (preferences) => {
    try {
        await AsyncStorage.setItem('userPreferences', JSON.stringify(preferences));
    } catch (error) {
        console.error("Error saving user preferences:", error);
    }
};

export const getUserPreferences = async () => {
    try {
        const preferences = await AsyncStorage.getItem('userPreferences');
        return preferences ? JSON.parse(preferences) : {};
    } catch (error) {
        console.error("Error retrieving user preferences:", error);
    }
};
